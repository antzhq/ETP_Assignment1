\section{Derived parameters}
\textbf{Efficiency chains}
\begin{align*}
\T{Eff}_{\T{optical}} &= \T{Eff}_{\T{reflectivity}}\cdot\T{Eff}_{\T{cosine}}\cdot\T{Eff}_{\T{shading blocking}}\cdot\T{Eff}_{\T{atmospheric}}\cdot\T{Eff}_{\T{spillage}}\\
\T{Eff}_{\T{CSP}} &= \T{Eff}_{\T{optical}}\cdot\T{Eff}_{\T{receiver}}\cdot\T{Eff}_{\T{rankine}}
\end{align*}
\textbf{Yield per m$^2$}, for all $\T{hr}\in\T{Hrs}$ and $\T{day}\in\T{Days}$ [W/m$^2$]
\begin{align*}
\T{Yield}_{\T{CSP elec,hr,day}} &= \T{Eff}_{\T{CSP}}\cdot\T{Irradiance}_{\T{hr,day}}\\
\T{Yield}_{\T{PVT elec,hr,day}} &= \T{Eff}_{\T{PVT,elec}}\cdot\T{Irradiance}_{\T{hr,day}}\\
\T{Yield}_{\T{PVT heat,hr,day}} &= \max\Big(0,\ \T{Factor}_{\T{heat removal}}\big[\T{Frac}_{\T{absorbed}}\cdot\T{Irradiance}_{\T{hr,day}}\\
&\hspace{6em}-\T{Coef}_{\T{heat loss}}\,(\T{T}_{\T{water in}}-\T{T}_{\T{ambient,hr,day}})\big]\Big)
\end{align*}
This is the heat-removal-factor form of the flat plate collector,
$\dot{Q}_u = F_R A_c\big[(\tau\alpha)_{\T{eff}} I_T - U(T_{\T{in}}-T_a)\big]$ with $A_r=A_c$,
divided by the collector area. Losses are evaluated at the inlet water temperature;
$\T{Factor}_{\T{heat removal}}$ corrects for the plate being warmer.

\textbf{PVT water outlet temperature}, for all $\T{hr}\in\T{Hrs}$ and $\T{day}\in\T{Days}$ [$^\circ$C]
\[
\T{T}_{\T{water out,hr,day}} = \T{T}_{\T{water in}} + \frac{\T{Area}_{\T{per PVT panel}}\cdot\T{Yield}_{\T{PVT heat,hr,day}}}{\T{FlowRate}_{\T{per PVT panel}}\cdot\T{c}_{\T{p,water}}}
\]

\section{Full model}
\textbf{Supply}, for all $\T{hr}\in\T{Hrs}$ and $\T{day}\in\T{Days}$ [GJ]
\begin{align*}
\T{Supply}_{\T{CSP elec,hr,day}} &= \T{GJ}_{\T{per W-hr}}\cdot\T{Area}_{\T{CSP}}\cdot\T{Yield}_{\T{CSP elec,hr,day}}\\
\T{Supply}_{\T{PVT elec,hr,day}} &= \T{GJ}_{\T{per W-hr}}\cdot\T{Area}_{\T{PVT}}\cdot\T{Yield}_{\T{PVT elec,hr,day}}\\
\T{Supply}_{\T{PVT heat,hr,day}} &= \T{GJ}_{\T{per W-hr}}\cdot\T{Area}_{\T{PVT}}\cdot\T{Yield}_{\T{PVT heat,hr,day}}
\end{align*}

\textbf{Objective function}
\[
\min\ \T{Cost}_{\T{total}}=\T{Cost}_{\T{CSP}}\cdot\T{Area}_{\T{CSP}}+\T{Cost}_{\T{PVT}}\cdot\T{Area}_{\T{PVT}}
\]

\textbf{Constraints}

Heat demand, for all $\T{day}\in\T{Days}$:
\begin{equation}\tag{C1}
\sum_{\T{hr}\in\T{SunHrs}_{\T{day}}}\T{Supply}_{\T{PVT heat,hr,day}}\ \ge\sum_{\T{hr}\in\T{SunHrs}_{\T{day}}}\T{Demand}_{\T{heat,hr,day}}
\end{equation}
Electricity demand, for all $\T{day}\in\T{Days}$:
\begin{equation}\tag{C2}
\sum_{\T{hr}\in\T{SunHrs}_{\T{day}}}\big(\T{Supply}_{\T{CSP elec,hr,day}}+\T{Supply}_{\T{PVT elec,hr,day}}\big)\ \ge\sum_{\T{hr}\in\T{SunHrs}_{\T{day}}}\T{Demand}_{\T{elec,hr,day}}
\end{equation}
Minimum CSP share, for all $\T{day}\in\T{Days}$:
\begin{multline}\tag{C3}
\sum_{\T{hr}\in\T{SunHrs}_{\T{day}}}\T{Supply}_{\T{CSP elec,hr,day}}\\
\ge\ \T{Share}_{\T{CSP,min}}\sum_{\T{hr}\in\T{SunHrs}_{\T{day}}}\big(\T{Supply}_{\T{CSP elec,hr,day}}+\T{Supply}_{\T{PVT elec,hr,day}}\big)
\end{multline}
Available area:
\begin{equation}\tag{C4}
\T{Area}_{\T{CSP}}+\T{Area}_{\T{PVT}}\le\T{Area}_{\T{max}}
\end{equation}
Integrality and non-negativity:
\begin{equation}\tag{C5}
\T{Count}_{\T{heliostats}},\ \T{Count}_{\T{PVT panels}}\in\{0,1,2,\dots\}
\end{equation}
